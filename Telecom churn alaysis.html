# 1. Store your 9 columns in a list
columns = ['PhoneService', 'MultipleLines', 'InternetService', 
           'OnlineSecurity', 'OnlineBackup', 'DeviceProtection', 
           'TechSupport', 'StreamingTV', 'StreamingMovies']

# 2. Create a 3x3 subplot canvas grid
fig, axes = plt.subplots(nrows=3, ncols=3, figsize=(16, 14))
axes = axes.flatten()  # Converts the 3x3 grid into a flat list of 9 positions

# 3. Loop through your columns and draw a countplot in each spot
for i, col in enumerate(columns):
    # Create the countplot for the current column
    sns.countplot(x=col, data=df, ax=axes[i],hue="Churn")
    
    # Simple one-liner to put number labels on every single bar
    [axes[i].bar_label(c) for c in axes[i].containers]
    
    # Set a clear title for each subplot
    axes[i].set_title(f'Count Plot of {col}')

# 4. Automatically adjust spacing so nothing overlaps
plt.tight_layout()
plt.show()
